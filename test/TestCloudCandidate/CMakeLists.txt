cmake_minimum_required(VERSION 3.16)
project(CloudCandidateTests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(Threads REQUIRED)
add_executable(cloud_core core.cpp)
target_include_directories(cloud_core PRIVATE ../../include)
target_link_libraries(cloud_core PRIVATE Threads::Threads)
if(WIN32)
  # Compile the real WinHTTP transport with Windows SDK, without Weasel PCH.
  target_sources(cloud_core PRIVATE ../../RimeWithWeasel/CloudCandidate.cpp)
  target_compile_definitions(cloud_core PRIVATE WEASEL_CLOUD_STANDALONE)
  target_link_libraries(cloud_core PRIVATE winhttp)
endif()
enable_testing()
add_test(NAME cloud_core COMMAND cloud_core)

if(MSVC)
  target_compile_options(cloud_core PRIVATE /utf-8 /EHsc)
endif()
